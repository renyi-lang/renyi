module lists.merging
  purpose: Combine sorted lists of Integers.

public function merge_sorted(left: List of Integer, right: List of Integer)
  returns List of Integer
  purpose: Merge two sorted lists of Integers into one sorted list without duplicates.
  example: merge_sorted(left: [1, 3, 5], right: [2, 4, 6]) is [1, 2, 3, 4, 5, 6]
  example: merge_sorted(left: [1, 2, 2, 5], right: [2, 3, 5, 8]) is [1, 2, 3, 5, 8]
  example: merge_sorted(left: [], right: [4, 7]) is [4, 7]
  example: merge_sorted(left: [], right: []) is []

  let combined be
    for each part in [left, right], entry in part
    sorted by entry
    collect entry
  let mutable merged: List of Integer be []
  for each candidate in combined
    let already_kept be for each kept in merged any kept is candidate
    if not already_kept then
      set merged to merged.append(candidate)
    end
  end
  return merged
end

test "merging two empty lists gives an empty list"
  check merge_sorted(left: [], right: []) is []
end
