module security.passwords
  purpose: Classify how strong a password is.

public type Strength is one of
  purpose: How hard a password is to guess.
  Weak
  Fair
  Strong
end

public function classify(password: Text) returns Strength
  purpose: Rate a password as Weak, Fair or Strong by length and character mix.
  example: classify("hello") is Weak
  example: classify("Ab1!xyz") is Weak
  example: classify("sunshine") is Fair
  example: classify("Tr0ub4dor&3") is Fair
  example: classify("abcdefghijkl") is Fair
  example: classify("correct-horse-battery") is Fair
  example: classify("Tr0ub4dor&3x") is Strong
  example: classify("blue-sky-2026") is Strong

  let has_strong_length be password.matches(raw "^.{12,}$")
  let has_fair_length be password.matches(raw "^.{8,}$")
  let has_letter be password.matches(raw "^.*[A-Za-z].*$")
  let has_digit be password.matches(raw "^.*[0-9].*$")
  let has_other_kind be password.matches(raw "^.*[^A-Za-z0-9].*$")

  if has_strong_length and has_letter and has_digit and has_other_kind then
    return Strong
  otherwise if has_fair_length then
    return Fair
  otherwise
    return Weak
  end
end
