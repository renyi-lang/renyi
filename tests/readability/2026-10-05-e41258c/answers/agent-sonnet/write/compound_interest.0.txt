module savings.schedule
  purpose: Print a yearly compound-interest schedule of a savings balance, exact to the cent.

import std.console
import std.environment
import std.json

public type AppError
  purpose: A problem with the command-line arguments.
  has detail: Text
  can ToText
end

function round_to_cents(amount: Decimal) returns Decimal
  purpose: Round a non-negative amount to the nearest cent, with halves rounding up.
  example: round_to_cents(2.344) is 2.34
  example: round_to_cents(2.345) is 2.35
  example: round_to_cents(0.00) is 0.00

  let shifted be amount + 0.005
  let leftover be shifted remainder 0.01
  return shifted - leftover
end

function interest_for(balance: Decimal, rate: Decimal) returns Decimal
  purpose: Compute one year of interest at a yearly rate in percent, rounded to the cent.
  example: interest_for(balance: 1000.00, rate: 5.00) is 50.00
  example: interest_for(balance: 10.50, rate: 5.00) is 0.53
  example: interest_for(balance: 250.00, rate: 3.50) is 8.75

  return round_to_cents(balance * rate / 100.00)
end

function argument_at(arguments: List of Text, position: Integer, name: Text)
  returns Text
  or fails with AppError
  purpose: Find the command-line argument at a zero-based position, or fail naming it.
  example: argument_at(arguments: ["1000", "5", "10"], position: 1, name: "rate") is "5"

  let mutable index be 0
  for each argument in arguments
    if index is position then
      return argument
    end
    set index to index + 1
  end
  fail with AppError(detail: "missing argument: {name}")
end

function read_decimal(arguments: List of Text, position: Integer, name: Text)
  returns Decimal
  or fails with AppError
  purpose: Read a non-negative decimal number from the command-line argument at a position.
  example: read_decimal(arguments: ["1000.50", "5"], position: 0, name: "amount") is 1000.50

  let text be argument_at(arguments: arguments, position: position, name: name) otherwise fail
  let number: Decimal be json.parse(text) otherwise fail with AppError(detail: "{name} is not a number")
  if number is less than 0.00 then
    fail with AppError(detail: "{name} must not be negative")
  end
  return number
end

function read_integer(arguments: List of Text, position: Integer, name: Text)
  returns Integer
  or fails with AppError
  purpose: Read a non-negative whole number from the command-line argument at a position.
  example: read_integer(arguments: ["1000", "5", "10"], position: 2, name: "years") is 10

  let text be argument_at(arguments: arguments, position: position, name: name) otherwise fail
  let number: Integer be json.parse(text) otherwise fail with AppError(detail: "{name} is not a whole number")
  if number is less than 0 then
    fail with AppError(detail: "{name} must not be negative")
  end
  return number
end

public function main() or fails with AppError
  needs console, environment
  purpose: Print the yearly schedule for the amount, rate and years given on the command line.

  let arguments be environment.arguments()
  let principal be read_decimal(arguments: arguments, position: 0, name: "amount") otherwise fail
  let rate be read_decimal(arguments: arguments, position: 1, name: "rate") otherwise fail
  let years be read_integer(arguments: arguments, position: 2, name: "years") otherwise fail

  console.print("Starting amount {principal}, yearly rate {rate} percent, {years} years")
  let mutable balance be principal
  for each year in from 1 to years
    let interest be interest_for(balance: balance, rate: rate)
    set balance to balance + interest
    console.print("Year {year}: interest {interest}, balance {balance}")
  end
end
