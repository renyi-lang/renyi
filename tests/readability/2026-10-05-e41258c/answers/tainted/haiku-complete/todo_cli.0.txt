public function parse_command(arguments: List of Text) returns Command
  purpose: Turn command-line words into a command; anything unrecognized is Help.
  example: parse_command(["add", "buy", "milk"]) is Add(task: "buy milk")
  example: parse_command(["done", "2"]) is Done(position: 2)
  example: parse_command([]) is Help

  let verb be arguments.first() otherwise return Help
  match verb
    when "add" then return Add(task: arguments.rest().join(" "))
    when "done" then
      let position_text be arguments.at(1) otherwise return Help
      let position be position_text.to_integer() otherwise return Help
      return Done(position: position) otherwise Help
    when "show" then return Show
    otherwise return Help
  end
end
