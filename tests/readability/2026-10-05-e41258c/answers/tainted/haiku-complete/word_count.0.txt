public function count_words(text: Text) returns List of WordCount
  purpose: Count every lower-cased word, most frequent first.
  example: count_words("a b a")
    is [WordCount(word: "a", occurrences: 2), WordCount(word: "b", occurrences: 1)]
  example: count_words("") is []

  let words be
    for each line in text.lines(), raw_word in line.split(" ")
    where raw_word.trim() is not ""
    collect raw_word.trim().to_lower()
  let grouped be for each word in words group by word
  return
    for each word, repeats in grouped
    sorted by repeats.length() descending
    collect WordCount(word: word, occurrences: repeats.length())
end
