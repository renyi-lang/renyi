public function quote(parcel: Parcel) returns Quote
  purpose: Price and delivery time for one parcel; the host calls this at every checkout.
  example: quote(Parcel(weight_grams: 500, destination: Domestic, order_total: 60.00))
    is Quote(cost: 0, business_days: 2, free_shipping: true)
  example: quote(Parcel(weight_grams: 2500, destination: Europe, order_total: 20.00))
    is Quote(cost: 14.50, business_days: 5, free_shipping: false)

  let is_free be
    match parcel.destination
      when Domestic then parcel.order_total is at least free_shipping_threshold
      otherwise false
    end
  end

  let cost be
    if is_free then 0.00
    otherwise base_cost(parcel.destination) + weight_steps(parcel.weight_grams).to_decimal() * 2.50
    end
  end

  return Quote(cost: cost, business_days: delivery_days(parcel.destination), free_shipping: is_free)
end
