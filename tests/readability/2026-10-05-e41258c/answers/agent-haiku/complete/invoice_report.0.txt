public function statement_line(order: Invoice) returns Text
  purpose: One row of the statement: customer, number of lines and total, in fixed columns.
  example: statement_line(Invoice(customer: "ACME", lines: [], discount_percent: 0))
    is "ACME          0 lines      0.00"

  let line_count be for each line in order.lines count
  let total be invoice.total(order)
  return "{order.customer}          {line_count} lines      {total}"
end
