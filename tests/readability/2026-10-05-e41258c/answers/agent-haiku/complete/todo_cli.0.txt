public function parse_command(arguments: List of Text) returns Command
  purpose: Turn command-line words into a command; anything unrecognized is Help.
  example: parse_command(["add", "buy", "milk"]) is Add(task: "buy milk")
  example: parse_command(["done", "2"]) is Done(position: 2)
  example: parse_command([]) is Help

  let command be
    for each arg in arguments first

  match command
    when some("add") then
      let rest be
        for each arg, index in arguments.with_index()
        where index is greater than 0
        collect arg
      let task be rest.join(" ")
      return Add(task: task)
    when some("done") then
      let second be
        for each arg, index in arguments.with_index()
        where index is 1
        first
      match second
        when some(position_text) then
          let position be position_text.to_integer() otherwise 1
          return Done(position: position)
        otherwise return Help
      end
    when some("show") then
      return Show
    otherwise
      return Help
  end
end
