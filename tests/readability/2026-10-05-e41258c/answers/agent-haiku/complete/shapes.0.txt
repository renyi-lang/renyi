public function describe(shape: Shape) returns Text
  purpose: One line naming the shape and giving its area.
  example: describe(Point) is "a point with area 0.00"
  example: describe(Rectangle(width: 2, height: 3)) is "a 2 by 3 rectangle with area 6.00"

  match shape
    when Circle(radius) then return "a circle of radius {radius} with area {shape.area()}"
    when Rectangle(width, height) then return "a {width} by {height} rectangle with area {shape.area()}"
    when Point then return "a point with area 0.00"
  end
end
