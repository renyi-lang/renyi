module password_strength
  purpose: Classify passwords as Weak, Fair, or Strong.

public type PasswordStrength is one of
  Weak
  Fair
  Strong
end

public function classify(password: Text) returns PasswordStrength
  purpose: Classify a password into Weak, Fair, or Strong based on its complexity.
  example: classify(password: "weak") is Weak
  example: classify(password: "MyPass12") is Fair
  example: classify(password: "MyStr0ng!Pass") is Strong

  if password.matches(raw "^.{12,}$") and
     password.matches(raw "[A-Za-z]") and
     password.matches(raw "[0-9]") and
     password.matches(raw "[^A-Za-z0-9]")
  then
    return Strong
  otherwise if password.matches(raw "^.{8,}$")
  then
    return Fair
  otherwise
    return Weak
  end
end
